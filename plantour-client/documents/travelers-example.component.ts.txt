import { Component, OnInit, OnDestroy } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ToolbarMenuService } from '../app/services/toolbar-menu.service';
import { MenuItem } from 'primeng/api';

@Component({
  selector: 'app-travelers',
  standalone: true,
  imports: [CommonModule],
  template: \`
    <h2>Travelers</h2>
    <p>Example child component that contributes its own toolbar menu.</p>
  \`
})
export class TravelersComponent implements OnInit, OnDestroy {
  constructor(private toolbarMenu: ToolbarMenuService) {}

  ngOnInit(): void {
    const items: MenuItem[] = [
      {
        label: 'Add traveler',
        icon: 'pi pi-user-plus',
        command: () => {
          console.log('Add traveler clicked');
        }
      },
      {
        label: 'Groups',
        icon: 'pi pi-users',
        command: () => {
          console.log('Groups clicked');
        }
      }
    ];

    this.toolbarMenu.setChildItems(items);
  }

  ngOnDestroy(): void {
    // Clear child menu when this component is destroyed
    this.toolbarMenu.clearChildItems();
  }
}
