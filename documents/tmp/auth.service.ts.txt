import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, BehaviorSubject, tap, map } from 'rxjs';
import { environment } from '../../environments/environment';
import { jwtDecode } from 'jwt-decode';
import {
  SignUpRequest,
  SignInRequest,
  SignUpParticipantRequest,
  SignInParticipantRequest,
  RefreshTokenRequest,
  AuthResponse,
  ParticipantAuthResponse,
  RefreshTokenResponse,
  ValidateTokenResponse,
  DecodedToken,
  CurrentUser
} from '../models/auth.models';

@Injectable({
  providedIn: 'root',
})
export class AuthService {
  private apiUrl = `${environment.apiUrl}/api/auth`;
  private readonly TOKEN_KEY = 'plantour_auth_token';
  private readonly REFRESH_TOKEN_KEY = 'plantour_refresh_token';
  
  private currentUserSubject = new BehaviorSubject<CurrentUser | null>(null);
  public currentUser$ = this.currentUserSubject.asObservable();

  constructor(private http: HttpClient) {
    this.initializeCurrentUser();
  }

  // ==================== ADMIN ENDPOINTS ====================

  /**
   * Admin registration
   */
  signUpAdmin(request: SignUpRequest): Observable<AuthResponse> {
    return this.http.post<AuthResponse>(`${this.apiUrl}/admin/signup`, request)
      .pipe(
        tap(response => this.handleAuthResponse(response, 'Admin'))
      );
  }

  /**
   * Admin sign in
   */
  signInAdmin(request: SignInRequest): Observable<AuthResponse> {
    return this.http.post<AuthResponse>(`${this.apiUrl}/admin/signin`, request)
      .pipe(
        tap(response => this.handleAuthResponse(response, 'Admin'))
      );
  }

  // ==================== PARTICIPANT ENDPOINTS ====================

  /**
   * Participant registration (requires admin authorization)
   */
  signUpParticipant(request: SignUpParticipantRequest): Observable<ParticipantAuthResponse> {
    return this.http.post<ParticipantAuthResponse>(`${this.apiUrl}/participant/signup`, request)
      .pipe(
        tap(response => this.handleParticipantAuthResponse(response))
      );
  }

  /**
   * Participant sign in by access code
   */
  signInParticipant(request: SignInParticipantRequest): Observable<ParticipantAuthResponse> {
    return this.http.post<ParticipantAuthResponse>(`${this.apiUrl}/participant/signin`, request)
      .pipe(
        tap(response => this.handleParticipantAuthResponse(response))
      );
  }

  // ==================== COMMON ENDPOINTS ====================

  /**
   * Refresh access token
   */
  refreshToken(refreshToken?: string): Observable<RefreshTokenResponse> {
    const token = refreshToken || this.getRefreshToken();
    if (!token) {
      throw new Error('Refresh token not found');
    }

    const request: RefreshTokenRequest = { refreshToken: token };
    return this.http.post<RefreshTokenResponse>(`${this.apiUrl}/refresh`, request)
      .pipe(
        tap(response => {
          this.setAccessToken(response.accessToken);
          this.setRefreshToken(response.refreshToken);
          this.updateCurrentUser();
        })
      );
  }

  /**
   * Revoke token (requires authorization)
   */
  revokeToken(refreshToken?: string): Observable<{ message: string }> {
    const token = refreshToken || this.getRefreshToken();
    if (!token) {
      throw new Error('Refresh token not found');
    }

    const request: RefreshTokenRequest = { refreshToken: token };
    return this.http.post<{ message: string }>(`${this.apiUrl}/revoke`, request)
      .pipe(
        tap(() => this.logout())
      );
  }

  /**
   * Validate token (requires authorization)
   */
  validateToken(): Observable<ValidateTokenResponse> {
    return this.http.get<ValidateTokenResponse>(`${this.apiUrl}/validate`);
  }

  // ==================== TOKEN MANAGEMENT ====================

  /**
   * Save access token
   */
  private setAccessToken(token: string): void {
    localStorage.setItem(this.TOKEN_KEY, token);
  }

  /**
   * Get access token
   */
  getAccessToken(): string | null {
    return localStorage.getItem(this.TOKEN_KEY);
  }

  /**
   * Save refresh token
   */
  private setRefreshToken(token: string): void {
    localStorage.setItem(this.REFRESH_TOKEN_KEY, token);
  }

  /**
   * Get refresh token
   */
  getRefreshToken(): string | null {
    return localStorage.getItem(this.REFRESH_TOKEN_KEY);
  }

  /**
   * Clear all tokens
   */
  private clearTokens(): void {
    localStorage.removeItem(this.TOKEN_KEY);
    localStorage.removeItem(this.REFRESH_TOKEN_KEY);
  }

  // ==================== CURRENT USER MANAGEMENT ====================

  /**
   * Initialize current user from localStorage
   */
  private initializeCurrentUser(): void {
    const token = this.getAccessToken();
    if (token && !this.isTokenExpired(token)) {
      this.updateCurrentUser();
    }
  }

  /**
   * Update current user information from token
   */
  private updateCurrentUser(): void {
    const user = this.getCurrentUserFromToken();
    this.currentUserSubject.next(user);
  }

  /**
   * Get current user from token
   */
  getCurrentUserFromToken(): CurrentUser | null {
    const token = this.getAccessToken();
    if (!token || this.isTokenExpired(token)) {
      return null;
    }

    try {
      const decoded = jwtDecode<DecodedToken>(token);
      
      // Determine role from token
      const role = decoded.role || (decoded.participantId ? 'Participant' : 'Admin');
      
      // Base fields
      const email = decoded.email || decoded.sub;
      const firstName = decoded.firstName || decoded.first_name;
      const lastName = decoded.lastName || decoded.last_name;
      
      // Build full name
      let fullName = '';
      if (firstName && lastName) {
        fullName = `${firstName} ${lastName}`;
      } else if (firstName || lastName) {
        fullName = firstName || lastName || '';
      } else {
        fullName = email;
      }

      const currentUser: CurrentUser = {
        id: decoded.userId || decoded.participantId || decoded.sub,
        email: email,
        firstName: firstName,
        lastName: lastName,
        fullName: fullName.trim(),
        role: role as 'Admin' | 'Participant'
      };

      // Additional fields for participant
      if (role === 'Participant') {
        currentUser.accessCode = decoded.accessCode;
        currentUser.adminId = decoded.adminId;
        currentUser.adminEmail = decoded.adminEmail;
      }

      return currentUser;
    } catch (error) {
      console.error('Error decoding token:', error);
      return null;
    }
  }

  /**
   * Get current user (synchronously)
   */
  getCurrentUser(): CurrentUser | null {
    return this.currentUserSubject.value;
  }

  /**
   * Get user full name
   */
  getUserFullName(): string {
    const user = this.getCurrentUser();
    return user?.fullName || 'Guest';
  }

  /**
   * Get user role
   */
  getUserRole(): 'Admin' | 'Participant' | null {
    const user = this.getCurrentUser();
    return user?.role || null;
  }

  /**
   * Check if user is admin
   */
  isAdmin(): boolean {
    return this.getUserRole() === 'Admin';
  }

  /**
   * Check if user is participant
   */
  isParticipant(): boolean {
    return this.getUserRole() === 'Participant';
  }

  // ==================== AUTHENTICATION STATUS ====================

  /**
   * Check if token is expired
   */
  isTokenExpired(token: string): boolean {
    if (!token) {
      return true;
    }

    try {
      const decoded = jwtDecode<DecodedToken>(token);
      const now = Math.floor(Date.now() / 1000);
      return decoded.exp < now;
    } catch (error) {
      return true;
    }
  }

  /**
   * Check if user is authenticated
   */
  isAuthenticated(): boolean {
    const token = this.getAccessToken();
    if (!token) {
      return false;
    }
    return !this.isTokenExpired(token);
  }

  /**
   * Logout from system
   */
  logout(): void {
    this.clearTokens();
    this.currentUserSubject.next(null);
  }

  // ==================== PRIVATE HELPERS ====================

  /**
   * Handle admin authentication response
   */
  private handleAuthResponse(response: AuthResponse, role: 'Admin' | 'Participant'): void {
    this.setAccessToken(response.accessToken);
    this.setRefreshToken(response.refreshToken);
    
    const currentUser: CurrentUser = {
      id: response.userId,
      email: response.email,
      firstName: response.firstName,
      lastName: response.lastName,
      fullName: this.buildFullName(response.firstName, response.lastName, response.email),
      role: role
    };
    
    this.currentUserSubject.next(currentUser);
  }

  /**
   * Handle participant authentication response
   */
  private handleParticipantAuthResponse(response: ParticipantAuthResponse): void {
    this.setAccessToken(response.accessToken);
    this.setRefreshToken(response.refreshToken);
    
    const currentUser: CurrentUser = {
      id: response.participantId,
      email: response.email,
      firstName: response.firstName,
      lastName: response.lastName,
      fullName: this.buildFullName(response.firstName, response.lastName, response.email),
      role: 'Participant',
      accessCode: response.accessCode,
      adminId: response.adminId,
      adminEmail: response.adminEmail
    };
    
    this.currentUserSubject.next(currentUser);
  }

  /**
   * Build full name from parts
   */
  private buildFullName(firstName?: string, lastName?: string, email?: string): string {
    if (firstName && lastName) {
      return `${firstName} ${lastName}`;
    }
    if (firstName || lastName) {
      return firstName || lastName || '';
    }
    return email || 'User';
  }
}
