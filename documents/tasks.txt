Stripe: # Creating a Production-Ready Stripe Integration for Subscription Management

To integrate Stripe with your .NET API + PostgreSQL + Angular application for subscription management, follow these comprehensive steps:

## Stripe Dashboard Setup

1. **Create a Stripe Account**
   - Sign up at [Stripe.com](https://stripe.com)
   - Complete the verification process

2. **Configure Your Account Settings**
   - Set business information, branding, and email templates
   - Configure [payment methods settings](https://dashboard.stripe.com/settings/payment_methods)

3. **Create Products and Pricing Plans**
   - Navigate to Products > Add Product
   - Create two products with monthly and yearly pricing options
   - Set appropriate billing cycles and prices

4. **Set Up Webhooks**
   - Go to Developers > Webhooks > Add Endpoint
   - Enter your API endpoint URL (e.g., `https://yourapi.com/api/webhooks/stripe`)
   - Select events to listen for (at minimum):
     - `customer.subscription.created`
     - `customer.subscription.updated`
     - `customer.subscription.deleted`
     - `invoice.payment_succeeded`
     - `invoice.payment_failed`
     - `invoice.upcoming`

5. **Get API Keys**
   - Go to Developers > API Keys
   - Save your publishable key and secret key

```sql
    email varchar(255) not null unique,
```

```csharp
public interface IStripeService
{
    Task<Customer> CreateOrUpdateCustomerAsync(string email, string name);
    Task<string> CreateCheckoutSessionAsync(Guid customerId, string priceId, string successUrl, string cancelUrl);
    Task<Subscription> CreateSubscriptionAsync(Guid customerId, string priceId);
    Task<Subscription> UpdateSubscriptionAsync(string subscriptionId, string newPriceId);
    Task CancelSubscriptionAsync(string subscriptionId, bool cancelAtPeriodEnd = true);
    Task<List<Stripe.Price>> GetPricesAsync();
}

public class StripeService : IStripeService
{
    private readonly CustomerService _customerService;
    private readonly SubscriptionService _subscriptionService;
    private readonly PaymentMethodService _paymentMethodService;
    private readonly PriceService _priceService;
    private readonly ApplicationDbContext _dbContext;

    public StripeService(
        CustomerService customerService,
        SubscriptionService subscriptionService,
        PaymentMethodService paymentMethodService,
        PriceService priceService,
        ApplicationDbContext dbContext)
    {
        _customerService = customerService;
        _subscriptionService = subscriptionService;
        _paymentMethodService = paymentMethodService;
        _priceService = priceService;
        _dbContext = dbContext;
    }

    public async Task<Customer> CreateOrUpdateCustomerAsync(string email, string name)
    {
        var customer = await _dbContext.Customers.FirstOrDefaultAsync(c => c.Email == email);
        
        if (customer == null)
        {
            // Create new customer in Stripe
            var stripeCustomer = await _customerService.CreateAsync(new CustomerCreateOptions
            {
                Email = email,
                Name = name
            });
            
            // Create new customer in database
            customer = new Customer
            {
                Email = email,
                Name = name,
                StripeCustomerId = stripeCustomer.Id,
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow
            };
            
            _dbContext.Customers.Add(customer);
            await _dbContext.SaveChangesAsync();
        }
        else if (customer.StripeCustomerId == null)
        {
            // Create Stripe customer for existing DB customer
            var stripeCustomer = await _customerService.CreateAsync(new CustomerCreateOptions
            {
                Email = email,
                Name = name
            });
            
            customer.StripeCustomerId = stripeCustomer.Id;
            customer.UpdatedAt = DateTime.UtcNow;
            await _dbContext.SaveChangesAsync();
        }
        
        return customer;
    }

    public async Task<string> CreateCheckoutSessionAsync(Guid customerId, string priceId, string successUrl, string cancelUrl)
    {
        var customer = await _dbContext.Customers.FindAsync(customerId);
        if (customer == null || string.IsNullOrEmpty(customer.StripeCustomerId))
            throw new Exception("Customer not found or not connected to Stripe");
            
        var options = new Stripe.Checkout.SessionCreateOptions
        {
            Customer = customer.StripeCustomerId,
            PaymentMethodTypes = new List<string> { "card" },
            LineItems = new List<Stripe.Checkout.SessionLineItemOptions>
            {
                new Stripe.Checkout.SessionLineItemOptions
                {
                    Price = priceId,
                    Quantity = 1
                }
            },
            Mode = "subscription",
            SuccessUrl = successUrl,
            CancelUrl = cancelUrl,
        };
        
        var service = new Stripe.Checkout.SessionService();
        var session = await service.CreateAsync(options);
        
        return session.Id;
    }

    public async Task<Subscription> CreateSubscriptionAsync(Guid customerId, string priceId)
    {
        var customer = await _dbContext.Customers.FindAsync(customerId);
        if (customer == null || string.IsNullOrEmpty(customer.StripeCustomerId))
            throw new Exception("Customer not found or not connected to Stripe");
            
        // Get default payment method
        var paymentMethod = await _dbContext.PaymentMethods
            .FirstOrDefaultAsync(p => p.CustomerId == customerId && p.IsDefault);
            
        if (paymentMethod == null)
            throw new Exception("No default payment method found");
            
        // Create subscription in Stripe
        var options = new SubscriptionCreateOptions
        {
            Customer = customer.StripeCustomerId,
            Items = new List<SubscriptionItemOptions>
            {
                new SubscriptionItemOptions
                {
                    Price = priceId,
                    Quantity = 1
                }
            },
            DefaultPaymentMethod = paymentMethod.StripePaymentMethodId,
        };
        
        var stripeSubscription = await _subscriptionService.CreateAsync(options);
        
        // Get price details
        var price = await _priceService.GetAsync(priceId);
        var product = await new ProductService().GetAsync(price.ProductId);
        
        // Create subscription in database
        var subscription = new Models.Subscription
        {
            CustomerId = customerId,
            StripeSubscriptionId = stripeSubscription.Id,
            StripePriceId = priceId,
            ProductName = product.Name,
            Status = stripeSubscription.Status,
            CurrentPeriodStart = stripeSubscription.CurrentPeriodStart,
            CurrentPeriodEnd = stripeSubscription.CurrentPeriodEnd,
            CancelAtPeriodEnd = stripeSubscription.CancelAtPeriodEnd,
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };
        
        _dbContext.Subscriptions.Add(subscription);
        await _dbContext.SaveChangesAsync();
        
        return subscription;
    }
```

```csharp
[ApiController]
[Route("api/webhooks")]
public class WebhooksController : ControllerBase
{
    private readonly IStripeWebhookService _webhookService;
    private readonly string _webhookSecret;
    private readonly ILogger<WebhooksController> _logger;

    public WebhooksController(
        IConfiguration configuration,
        IStripeWebhookService webhookService,
        ILogger<WebhooksController> logger)
    {
        _webhookSecret = configuration["Stripe:WebhookSecret"];
        _webhookService = webhookService;
        _logger = logger;
    }

    [HttpPost("stripe")]
    public async Task<IActionResult> StripeWebhook()
    {
        var json = await new StreamReader(HttpContext.Request.Body).ReadToEndAsync();
        
        try
        {
            var stripeEvent = EventUtility.ConstructEvent(
                json,
                Request.Headers["Stripe-Signature"],
                _webhookSecret
            );
            
            // Check if we've already processed this event
            var isProcessed = await _webhookService.IsEventProcessed(stripeEvent.Id);
            if (isProcessed)
            {
                return Ok();
            }
            
            // Handle the event based on its type
            switch (stripeEvent.Type)
            {
                case Events.CustomerSubscriptionCreated:
                    await _webhookService.HandleSubscriptionCreatedAsync(stripeEvent);
                    break;
                case Events.CustomerSubscriptionUpdated:
                    await _webhookService.HandleSubscriptionUpdatedAsync(stripeEvent);
                    break;
                case Events.CustomerSubscriptionDeleted:
                    await _webhookService.HandleSubscriptionDeletedAsync(stripeEvent);
                    break;
                case Events.InvoicePaymentSucceeded:
                    await _webhookService.HandleInvoicePaymentSucceededAsync(stripeEvent);
                    break;
                case Events.InvoicePaymentFailed:
                    await _webhookService.HandleInvoicePaymentFailedAsync(stripeEvent);
                    break;
                // Handle other relevant events
            }
            
            // Mark event as processed
            await _webhookService.MarkEventAsProcessed(stripeEvent.Id, stripeEvent.Type);
            
            return Ok();
        }
        catch (StripeException e)
        {
            _logger.LogError(e, "Error processing Stripe webhook");
            return BadRequest();
        }
    }
}

public interface IStripeWebhookService
{
    Task<bool> IsEventProcessed(string eventId);
    Task MarkEventAsProcessed(string eventId, string eventType);
    Task HandleSubscriptionCreatedAsync(Event stripeEvent);
    Task HandleSubscriptionUpdatedAsync(Event stripeEvent);
    Task HandleSubscriptionDeletedAsync(Event stripeEvent);
    Task HandleInvoicePaymentSucceededAsync(Event stripeEvent);
    Task HandleInvoicePaymentFailedAsync(Event stripeEvent);
}

public class StripeWebhookService : IStripeWebhookService
{
    private readonly ApplicationDbContext _dbContext;
    private readonly ILogger<StripeWebhookService> _logger;

    public StripeWebhookService(
        ApplicationDbContext dbContext,
        ILogger<StripeWebhookService> logger)
    {
        _dbContext = dbContext;
        _logger = logger;
    }

    public async Task<bool> IsEventProcessed(string eventId)
    {
        return await _dbContext.WebhookEvents.AnyAsync(e => e.StripeEventId == eventId);
    }

    public async Task MarkEventAsProcessed(string eventId, string eventType)
    {
        _dbContext.WebhookEvents.Add(new WebhookEvent
        {
            StripeEventId = eventId,
            StripeEventType = eventType,
            ProcessedAt = DateTime.UtcNow
        });
        
        await _dbContext.SaveChangesAsync();
    }

    public async Task HandleSubscriptionCreatedAsync(Event stripeEvent)
    {
        var subscription = stripeEvent.Data.Object as Stripe.Subscription;
        if (subscription == null) return;
        
        var customer = await _dbContext.Customers.FirstOrDefaultAsync(c => c.StripeCustomerId == subscription.CustomerId);
        if (customer == null) return;
        
        // Get price details
        var priceId = subscription.Items.Data[0].Price.Id;
        var price = await new PriceService().GetAsync(priceId);
        var product = await new ProductService().GetAsync(price.ProductId);
        
        // Create or update subscription in database
        var existingSubscription = await _dbContext.Subscriptions
            .FirstOrDefaultAsync(s => s.StripeSubscriptionId == subscription.Id);
            
        if (existingSubscription == null)
        {
            _dbContext.Subscriptions.Add(new Models.Subscription
            {
                CustomerId = customer.Id,
                StripeSubscriptionId = subscription.Id,
                StripePriceId = priceId,
                ProductName = product.Name,
                Status = subscription.Status,
                CurrentPeriodStart = subscription.CurrentPeriodStart,
                CurrentPeriodEnd = subscription.CurrentPeriodEnd,
                CancelAtPeriodEnd = subscription.CancelAtPeriodEnd,
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow
            });
        }
        
        await _dbContext.SaveChangesAsync();
    }
```

```csharp
[ApiController]
[Route("api/subscriptions")]
[Authorize]
public class SubscriptionsController : ControllerBase
{
    private readonly IStripeService _stripeService;
    private readonly ApplicationDbContext _dbContext;

    public SubscriptionsController(
        IStripeService stripeService,
        ApplicationDbContext dbContext)
    {
        _stripeService = stripeService;
        _dbContext = dbContext;
    }

    [HttpGet]
    public async Task<IActionResult> GetSubscriptions()
    {
        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        var customer = await _dbContext.Customers.FirstOrDefaultAsync(c => c.UserId == userId);
        
        if (customer == null)
            return NotFound("Customer not found");
            
        var subscriptions = await _dbContext.Subscriptions
            .Where(s => s.CustomerId == customer.Id)
            .ToListAsync();
            
        return Ok(subscriptions);
    }

    [HttpPost("checkout")]
    public async Task<IActionResult> CreateCheckoutSession(CheckoutSessionRequest request)
    {
        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        var customer = await _dbContext.Customers.FirstOrDefaultAsync(c => c.UserId == userId);
        
        if (customer == null)
            return NotFound("Customer not found");
            
        var sessionId = await _stripeService.CreateCheckoutSessionAsync(
            customer.Id,
            request.PriceId,
            request.SuccessUrl,
            request.CancelUrl
        );
        
        return Ok(new { SessionId = sessionId });
    }

    [HttpPost("{subscriptionId}/cancel")]
    public async Task<IActionResult> CancelSubscription(string subscriptionId, [FromQuery] bool atPeriodEnd = true)
    {
        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        var customer = await _dbContext.Customers.FirstOrDefaultAsync(c => c.UserId == userId);
        
        if (customer == null)
            return NotFound("Customer not found");
            
        var subscription = await _dbContext.Subscriptions
            .FirstOrDefaultAsync(s => s.StripeSubscriptionId == subscriptionId && s.CustomerId == customer.Id);
            
        if (subscription == null)
            return NotFound("Subscription not found");
            
        await _stripeService.CancelSubscriptionAsync(subscriptionId, atPeriodEnd);
        
        return Ok();
    }

    // Additional endpoints for updating subscriptions, viewing invoices, etc.
}

public class CheckoutSessionRequest
{
    public string PriceId { get; set; }
    public string SuccessUrl { get; set; }
    public string CancelUrl { get; set; }
```
